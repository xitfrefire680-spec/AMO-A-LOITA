local function fmtValue(v)
    if not v or v <= 0 then return "?" end
    if v >= 1e12 then return string.format("%.2fT", v/1e12) end
    if v >= 1e9  then return string.format("%.2fB", v/1e9)  end
    if v >= 1e6  then return string.format("%.2fM", v/1e6)  end
    if v >= 1e3  then return string.format("%.2fK", v/1e3)  end
    return string.format("%.0f", v)
end

local function pickIcon(tbl)
    if type(tbl) ~= "table" then return nil end
    local keys = { "Icon","Image","Thumbnail","IconId","ImageId","Texture","Asset","DisplayImage" }
    for _, k in ipairs(keys) do
        local v = tbl[k]
        if type(v) == "string" and v ~= "" then
            return v:match("^%d+$") and ("rbxassetid://"..v) or v
        elseif type(v) == "number" and v > 0 then
            return "rbxassetid://"..v
        end
    end
    return nil
end

local function NeutralizeTraps()

    if not d then return end

local function GroundMove(targetXZ, speed, easeOut)
    local _, hrp = getChar()
    if not hrp or not targetXZ then return false end
    local groundY = hrp.Position.Y
    local goal = Vector3.new(targetXZ.X, groundY, targetXZ.Z)
    local start = hrp.Position
    local dx, dz = goal.X - start.X, goal.Z - start.Z
    local dist = math.sqrt(dx*dx + dz*dz)
    if dist < 1 then
        hrp.CFrame = CFrame.new(goal)
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
        return true
    end
    speed = math.clamp(tonumber(speed) or 300, 50, 40000)
    local t0 = os.clock()
    local maxTime = (dist / math.max(speed, 50)) * 2 + 4
    local cx, cz = hrp.Position.X, hrp.Position.Z
    while not HUB.dead do
        local dt = RunService.Heartbeat:Wait()
        local tx, tz = goal.X - cx, goal.Z - cz
        local rem = math.sqrt(tx*tx + tz*tz)
        if rem < 1 then break end
        local step = math.min(speed * dt, rem)
        local ux, uz = tx / rem, tz / rem
        cx = cx + ux * step cz = cz + uz * step
        local np = Vector3.new(cx, groundY, cz)
        hrp.CFrame = CFrame.lookAt(np, Vector3.new(cx + ux, groundY, cz + uz))
        local realVel = step / math.max(dt, 0.001)
        hrp.AssemblyLinearVelocity = Vector3.new(ux * realVel, 0, uz * realVel)
        hrp.AssemblyAngularVelocity = Vector3.zero
        SetSpoofWalkSpeed(HUB.flySpeed)
        if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
        SyncHoverPad()
        if os.clock() - t0 > maxTime then break end
    end
    hrp.CFrame = CFrame.new(goal)
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    return true
end

local function TravelRoad(target, spd, approach)
    local _, hrp = getChar()
    if not hrp or not target then return false end
    pcall(NeutralizeTraps)
    local sp0 = hrp.Position
    local retBase = (target.X < 560)
    if retBase and sp0.X > SAFE_X then
        GroundMove(Vector3.new(sp0.X, 0, ROAD_Z), spd, false)
        GroundMove(Vector3.new(SAFE_X, 0, ROAD_Z), spd, false)
        GroundMove(Vector3.new(target.X, 0, ROAD_Z), SAFE_SPD, false)
        GroundMove(Vector3.new(target.X, 0, target.Z), SAFE_SPD, approach == true)
    else
        GroundMove(Vector3.new(sp0.X, 0, ROAD_Z), spd, false)
        GroundMove(Vector3.new(target.X, 0, ROAD_Z), spd, false)
        GroundMove(Vector3.new(target.X, 0, target.Z), spd, approach == true)
    end
    return true
end
local function GroundRush(target, spd, approach)
    local _, hrp = getChar()
    if not hrp or not target then return false end
    pcall(NeutralizeTraps)
    GroundMove(Vector3.new(target.X, 0, target.Z), spd, approach == true)
    return true
end
local function TravelWalk(target)
    local h, r = getChar()
    if not h or not r or not target then return false end
    pcall(NeutralizeTraps)
    local gy = r.Position.Y
    for _, pt in ipairs({
        Vector3.new(r.Position.X, gy, ROAD_Z),
        Vector3.new(target.X, gy, ROAD_Z),
        Vector3.new(target.X, gy, target.Z),
    }) do
        if HUB.dead then break end
        h:MoveTo(pt)
        local t0 = os.clock()
        while (r.Position - pt).Magnitude > 4.5 and os.clock() - t0 < 5 and not HUB.dead do task.wait(0.05) end
    end
    return true
end

    if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * 3 * HEADROOM) end

end

local function flyTo(targetPos)
    if typeof(targetPos) ~= "Vector3" then return false end
    local m = HUB.flyMode
    if m == "Safe Walk"   then return TravelWalk(targetPos) end
    if m == "Road Path"   then return TravelRoad(targetPos, HUB.flySpeed, true) end
    if m == "Admin Abuse" then return AdminAbuseTP(targetPos) end
    return GroundRush(targetPos, HUB.flySpeed, true)
end

        local _, r = getChar() if not r then return end
        local cHrp = carrier and getPlayerHrp(carrier) if not cHrp then return end

        r.AssemblyLinearVelocity = Vector3.zero r.AssemblyAngularVelocity = Vector3.zero
    end)
    while tick() - t0 < 90 and not HUB.dead do
        if not HUB.running and not (HUB.activeTarget and HUB.activeTarget.uid == uid) then
            stickConn:Disconnect() return false
        end
        local rec = findRecord(uid)
        if not rec then stickConn:Disconnect() return false end
        if rec.State ~= "Carried" then stickConn:Disconnect() return stealEgg(rec) end
        if tick() - lastCheck > 1 then
            lastCheck = tick()
            local nc = getPlayerByUserId(rec.CarrierUserId or carrierUserId)
            if nc then carrier = nc end
            if carrier and not carrier.Parent then stickConn:Disconnect() return false end
        end
        local weapon = getEquippedWeapon() if weapon then pcall(function() weapon:Activate() end) end
        local _, curHrp = getChar()
        if curHrp then
            for _, d in ipairs(Workspace:GetDescendants()) do
                if d:IsA("ProximityPrompt") and d.Enabled then
                    local nm = d.Name

                            pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end)
                        end
                    end
                end
            end
        end
        task.wait(0.05)
    end
    stickConn:Disconnect()
    return false
end

stealEgg = function(record)

    local uid = record.Uid if not uid then return false end
    if record.State == "Carried" and record.CarrierUserId and record.CarrierUserId ~= LP.UserId then
        return trackAndSteal(record.CarrierUserId, uid)
    end
    local targetPos = record.BoundsCFrame and record.BoundsCFrame.Position or record.Position or (record.CFrame and record.CFrame.Position)
    if not targetPos then return false end
    flyTo(Vector3.new(targetPos.X, 0, targetPos.Z))
    if HUB.dead then return false end
    local _, hrp = getChar() if not hrp then return false end
    local slotKey = nil
    if AreaEggSlotId and AreaEggSlotId.LooksLikeFirstAreaUid and AreaEggSlotId.LooksLikeFirstAreaUid(uid) then

    end
    if EggWorld and EggWorld.AskFieldEggCarry then
        pcall(function() EggWorld.AskFieldEggCarry:InvokeServer({ Uid=uid, FirstAreaSlotKey=slotKey }) end)
    end
    pcall(function() if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end end)
    local function tryPrompt()
        local _, curHrp = getChar() if not curHrp then return false end
        for _, d in ipairs(Workspace:GetDescendants()) do
            if d:IsA("ProximityPrompt") and d.Enabled then
                local nm = d.Name
                if nm == "CarryAreaEgg" or nm == "PickupEgg" or nm == "CarryEgg" or nm == "GrabEgg" or nm == "Pickup" then
                    local p = d.Parent if p and p:IsA("Attachment") then p = p.Parent end
                    if p and p:IsA("BasePart") and (p.Position - curHrp.Position).Magnitude < 18 then
                        pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end)
                        return true
                    end
                end
            end
        end
        return false
    end
    tryPrompt()
    local t0 = tick()
    while tick() - t0 < 3 and not HUB.dead do
        if isCarryingEgg() then return true end
        local curRec = findRecord(uid)
        if curRec and curRec.State == "Carried" and curRec.CarrierUserId and curRec.CarrierUserId ~= LP.UserId then

        end
        pcall(function() if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end end)
        tryPrompt()
        task.wait(0.1)
    end
    return isCarryingEgg()

        if sp:IsA("BasePart") then return sp.Position end
        if sp:IsA("Attachment") then return sp.WorldPosition end
        if sp:IsA("Model") then
            local prim = sp.PrimaryPart or sp:FindFirstChildWhichIsA("BasePart")
            if prim then return prim.Position end
        end
    end
    if plot:IsA("Model") then
        local prim = plot.PrimaryPart or plot:FindFirstChildWhichIsA("BasePart")
        if prim then return prim.Position end
    elseif plot:IsA("BasePart") then return plot.Position end
    return default
end
local function plantAllEggs()
    if not EggState or not EggState.PlantEgg or not EggToolDisplay then return 0 end
    local ids = {}
    for _, container in ipairs({LP.Character, LP:FindFirstChild("Backpack")}) do
        if container then
            for _, t in ipairs(container:GetChildren()) do
                if t:IsA("Tool") and EggToolDisplay.IsEggTool and EggToolDisplay.IsEggTool(t) then
                    local uid = EggToolDisplay.GetToolUid and EggToolDisplay.GetToolUid(t)
                    if uid then table.insert(ids, uid) end
                end
            end
        end
    end
    local count = 0
    for _, uid in ipairs(ids) do
        local ok, res = pcall(function() return EggState.PlantEgg(uid, CFrame.new(math.random(-6,6), 0, math.random(-6,6))) end)
        if ok and res then count = count + 1 end
        task.wait(0.1)
    end
    return count
end

local function returnHome(flyBackPos)
    if not isCarryingEgg() then return false end
    local dest = getPlot1Pos()
    local targetPos = Vector3.new(dest.X, dest.Y + 2, dest.Z)
    local _, r1 = getChar()
    if r1 then
        r1.CFrame = CFrame.new(targetPos)
        r1.AssemblyLinearVelocity = Vector3.zero
        r1.AssemblyAngularVelocity = Vector3.zero
    end
    SetSpoofWalkSpeed(HUB.flySpeed)
    if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
    SyncHoverPad()

    local savedDead = HUB.dead
    HUB.dead = false
    pcall(function() GroundMove(Vector3.new(dest.X, 0, dest.Z), HUB.flySpeed, true) end)
    HUB.dead = savedDead

    for _ = 1, 5 do
        plantAllEggs()
        task.wait(0.12)
        if not isCarryingEgg() then break end
    end

    if flyBackPos and typeof(flyBackPos) == "Vector3" then
        local savedDead2 = HUB.dead
        HUB.dead = false
        pcall(function() GroundMove(Vector3.new(flyBackPos.X, 0, flyBackPos.Z), HUB.flySpeed, true) end)
        HUB.dead = savedDead2
    end
    return true
end

local ChickenBossFarm = {
    enabled = false, stage = "idle",
    altNames = {"galinha","hen","rooster","chicken"},
    hitThreshold = 2, lastHealth = 100,
    stageStartTime = 0, waitTimeout = 15,
}
local function findChicken()
    local hints = {}
    for _, n in ipairs(ChickenBossFarm.altNames) do table.insert(hints, n:lower()) end
    local candidates = {}
    local function scan(container, depth)
        if depth > 6 or not container then return end
        for _, obj in ipairs(container:GetChildren()) do
            if obj:IsA("Model") or obj:IsA("BasePart") then
                local n = obj.Name:lower()
                for _, hint in ipairs(hints) do
                    if n:find(hint, 1, true) then
                        local pos
                        if obj:IsA("BasePart") then pos = obj.Position
                        else
                            local prim = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                            if prim then pos = prim.Position end
                        end
                        if pos then table.insert(candidates, {obj=obj, pos=pos}) end
                        break
                    end
                end
            end
            if obj:IsA("Model") or obj:IsA("Folder") then scan(obj, depth + 1) end
        end
    end
    scan(workspace, 0)
    local _, hrp = getChar()
    if hrp and #candidates > 1 then
        table.sort(candidates, function(a,b) return (a.pos-hrp.Position).Magnitude < (b.pos-hrp.Position).Magnitude end)
    end
    return candidates[1]
end
local function grabChickenEgg(chickenObj)
    if not chickenObj then return false end
    for _, d in ipairs(chickenObj:GetDescendants()) do
        if d:IsA("ProximityPrompt") and d.Enabled then
            pcall(function() d.HoldDuration = 0 fireproximityprompt(d) end)
        end
    end
    local _, hrp = getChar()
    if hrp then
        local anchor = chickenObj:IsA("BasePart") and chickenObj or chickenObj:FindFirstChildWhichIsA("BasePart")
        if anchor then pcall(function() firetouchinterest(hrp, anchor, 0) firetouchinterest(hrp, anchor, 1) end) end
    end
    return true
end
local function RunChickenSequence()
    if not ChickenBossFarm.enabled or ChickenBossFarm.stage == "done" then return end
    if HUB.dead then return end
    if ChickenBossFarm.stage == "idle" then
        ChickenBossFarm.stage = "going_chicken"
        local chicken = findChicken()
        if not chicken then
            ChickenBossFarm.stage = "done" return
        end
        flyTo(chicken.pos)
        local chicken2 = findChicken()
        if chicken2 then grabChickenEgg(chicken2.obj) end
        ChickenBossFarm.lastHealth = (getChar() and (getChar().Health or 100)) or 100
        ChickenBossFarm.stage = "waiting_hit"
        ChickenBossFarm.stageStartTime = tick()
        return
    end
    if ChickenBossFarm.stage == "waiting_hit" then
        if tick() - ChickenBossFarm.stageStartTime > ChickenBossFarm.waitTimeout then
            ChickenBossFarm.stage = "done"
        end
    end
end
RunService.Heartbeat:Connect(function()
    if not ChickenBossFarm.enabled or ChickenBossFarm.stage ~= "waiting_hit" then return end
    local hum = getChar() if not hum then return end
    if hum.Health < ChickenBossFarm.lastHealth - ChickenBossFarm.hitThreshold then
        ChickenBossFarm.stage = "done"
        ChickenBossFarm.enabled = false
    end
    ChickenBossFarm.lastHealth = hum.Health
end)

local AutoHitSystem = { lastHit = 0, cooldown = 0.35, busy = false }
local function findEnemyCarrier()
    if not EggState or not EggState.ReadFieldEggs then return nil end
    local ok, snap = pcall(EggState.ReadFieldEggs)
    if not ok or not snap or not snap.Records then return nil end
    local best, bestVal = nil, -1
    for _, r in ipairs(snap.Records) do
        if r.State == "Carried" and r.CarrierUserId and r.CarrierUserId ~= LP.UserId then
            local carrier = getPlayerByUserId(r.CarrierUserId)
            if carrier and getPlayerHrp(carrier) then
                local cat = r.AssetCategory or r.Category
                local val = getPetValue(cat)
                if val > bestVal then
                    bestVal = val
                    best = { record = r, carrier = carrier, uid = r.Uid, value = val }
                end
            end
        end
    end
    return best
end
local function equipBestWeapon()
    local char = LP.Character if not char then return nil end
    local isWeapon = function(tool)
        local n = string.lower(tool.Name)
        return n:find("bat") or n:find("sword") or n:find("hammer") or n:find("axe") or n:find("blade") or n:find("staff") or n:find("club") or n:find("katana")
    end
    local equipped = char:FindFirstChildOfClass("Tool")
    if equipped and isWeapon(equipped) then return equipped end
    local bp = LP:FindFirstChildOfClass("Backpack") if not bp then return nil end
    for _, t in ipairs(bp:GetChildren()) do
        if t:IsA("Tool") and isWeapon(t) then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                pcall(function() hum:EquipTool(t) end)
                task.wait(0.08)
                local eq = char:FindFirstChildOfClass("Tool")
                if eq then return eq end
            end
        end
    end
    return nil
end
local function runAutoHit()
    if AutoHitSystem.busy or HUB.dead or not HUB.hitEnabled then return end
    if os.clock() - AutoHitSystem.lastHit < AutoHitSystem.cooldown then return end
    local target = findEnemyCarrier()
    if not target then return end
    AutoHitSystem.busy = true
    local cHrp = getPlayerHrp(target.carrier)
    local _, myHrp = getChar()
    if not cHrp or not myHrp then AutoHitSystem.busy = false return end
    local dist = (myHrp.Position - cHrp.Position).Magnitude
    if dist > 12 then
        local moveTo = cHrp.Position + cHrp.CFrame.LookVector * 3 + Vector3.new(0, 2, 0)
        myHrp.CFrame = CFrame.new(moveTo)
        myHrp.AssemblyLinearVelocity = Vector3.zero
        myHrp.AssemblyAngularVelocity = Vector3.zero
        SetSpoofWalkSpeed(HUB.flySpeed)
        if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
        SyncHoverPad()
        task.wait(0.08)
    end
    pcall(function()
        myHrp.CFrame = CFrame.lookAt(myHrp.Position, Vector3.new(cHrp.Position.X, myHrp.Position.Y, cHrp.Position.Z))
    end)
    local weapon = equipBestWeapon()
    if weapon then
        pcall(function() weapon:Activate() end)
        AutoHitSystem.lastHit = os.clock()
    end
    AutoHitSystem.busy = false
end
task.spawn(function()
    while PGui:FindFirstChild("BlackHub") do
        RunService.Heartbeat:Wait()
        if HUB.hitEnabled then pcall(runAutoHit) end
    end
end)

local function findForestEgg()
    local eggs = getFieldEggs()
    if AreaEggSlotId and AreaEggSlotId.LooksLikeFirstAreaUid then
        for _, e in ipairs(eggs) do
            if AreaEggSlotId.LooksLikeFirstAreaUid(e.uid) then
                return e
            end
        end
    end
    for _, e in ipairs(eggs) do
        local a = string.lower(tostring(e.area or ""))
        if a == "forest" or a:find("floresta") then
            return e
        end
    end
    return nil
end

local function dropCarriedEgg()
    if not isCarryingEgg() then return true end
    if EggWorld then
        local drop = EggWorld.AskFieldEggDrop
        if drop and drop.InvokeServer then
            pcall(function() drop:InvokeServer({ Reason = "PlayerRequest" }) end)
            task.wait(0.3)
            if not isCarryingEgg() then return true end
        end
    end
    if EggState and EggState.DropFieldEgg and EggState.ReadFieldEggs then
        local ok, snap = pcall(EggState.ReadFieldEggs)
        if ok and snap and snap.Records then
            for _, r in ipairs(snap.Records) do
                if r.State == "Carried" and r.CarrierUserId == LP.UserId then
                    pcall(function() EggState.DropFieldEgg(r.Uid) end)
                    break
                end
            end
        end
        task.wait(0.3)
    end
    return not isCarryingEgg()
end

local function runPreSteal()
    local forestEgg = findForestEgg()
    if not forestEgg then return false end
    if not flyTo(Vector3.new(forestEgg.pos.X, 0, forestEgg.pos.Z)) then return false end
    if HUB.dead then return false end

    local uid = forestEgg.uid
    local rec = findRecord(uid) or forestEgg.record
    local slotKey = nil
    if AreaEggSlotId and AreaEggSlotId.LooksLikeFirstAreaUid
       and AreaEggSlotId.LooksLikeFirstAreaUid(uid) and rec then
        slotKey = AreaEggSlotId.SlotKey(rec.AreaId, rec.NestId)
    end

    if EggWorld and EggWorld.AskFieldEggCarry then
        pcall(function()
            EggWorld.AskFieldEggCarry:InvokeServer({ Uid = uid, FirstAreaSlotKey = slotKey })
        end)
    end
    pcall(function()
        if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end
    end)

    local t0 = tick()
    while tick() - t0 < 3 and not HUB.dead do
        if isCarryingEgg() then break end
        pcall(function()
            if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end
        end)
        task.wait(0.1)
    end
    if not isCarryingEgg() then return false end

    dropCarriedEgg()
    return true
end

local function runCycle()
    if HUB.busy or HUB.dead or not HUB.running then return false end
    HUB.busy = true

    if not HUB.preStealDone then
        local ok = runPreSteal()
        HUB.preStealDone = ok
        HUB.busy = false
        if ok then
            showToast("🌲 Floresta OK · indo pro real egg")
        end
        return ok
    end

    local origSpeed = HUB.flySpeed
    HUB.flySpeed = HUB.highSpeed or 800

    local result = false

    if isCarryingEgg() then
        returnHome()
        result = true

    elseif HUB.activeTarget and HUB.activeTarget.uid then
        local rec = findRecord(HUB.activeTarget.uid)
        if rec then
            local _, hrpBefore = getChar()
            local flyPos = hrpBefore and hrpBefore.Position
            if stealEgg(rec) then
                HUB.activeTarget = nil
                returnHome(flyPos)
                result = true
            end
        else
            HUB.activeTarget = nil
        end

    else
        local eggs = getFieldEggs()
        if #eggs > 0 then
            local target = nil
            if HUB.selectedUid then
                for _, e in ipairs(eggs) do
                    if e.uid == HUB.selectedUid then target = e break end
                end
            end
            if not target then
                table.sort(eggs, function(a, b)
                    if a.value ~= b.value then return a.value > b.value end
                    return a.rank > b.rank
                end)
                target = eggs[1]
            end

            local _, hrpB = getChar()
            local flyPosB = hrpB and hrpB.Position
            HUB.activeTarget = { uid = target.uid, name = target.name }

            if stealEgg(target.record) then
                HUB.activeTarget = nil
                returnHome(flyPosB)
                result = true
            end
        end
    end

    HUB.flySpeed = origSpeed
    HUB.busy = false
    return result
end

-- ============ HELPERS UI ============
local function getNormalizedRarity(name)
    return string.lower(tostring(name or "")):gsub("%s+", "")
end
local function isRarityFilterActive()
    for _, v in ipairs(RARITY_LIST) do
        if U.rarityFilter[v.Key] then return true end
    end
    return false
end
local function eggMatchesFilter(egg)
    if not isRarityFilterActive() then return true end
    return U.rarityFilter[getNormalizedRarity(egg.rarity)] == true
end
local function getFilteredEggs()
    local eggs = getFieldEggs()
    local out = {}
    for _, e in ipairs(eggs) do
        if eggMatchesFilter(e) then out[#out+1] = e end
    end
    table.sort(out, function(a, b)
        if a.value ~= b.value then return a.value > b.value end
        return a.rank > b.rank
    end)
    return out
end

local function updateRarityChecks()
    for _, cfg in ipairs(RARITY_LIST) do
        local ui = U.rarityChecks[cfg.Key]
        local on = U.rarityFilter[cfg.Key] == true
        ui.Box.Text = on and "X" or ""
        ui.Box.BackgroundColor3 = cfg.Color
        ui.Box.BackgroundTransparency = on and 0 or 0.55
        ui.Box.TextColor3 = on and Color3.fromRGB(8, 8, 10) or cfg.Color
        ui.Stroke.Transparency = on and 0 or 0.35
        ui.Row.BackgroundColor3 = on and C.cardSel or C.card
    end
end

local function updateBestDisplay()
    if not frame.Visible then return end
    local eggs = getFilteredEggs()
    local target = nil
    if HUB.selectedUid then
        for _, e in ipairs(eggs) do
            if e.uid == HUB.selectedUid then target = e break end
        end
    end
    if not target and #eggs > 0 then target = eggs[1] end
    U.bestEgg = target
    if not target then
        if bestName.Text ~= "NO EGG FOUND" then bestName.Text = "NO EGG FOUND" end
        if bestRarity.Text ~= "" then bestRarity.Text = "" end
        if bestValue.Text ~= "--" then bestValue.Text = "--" end
        previewImg.Image = ""
        previewImg.Visible = false
        previewEmoji.Visible = true
        return
    end
    local valStr = fmtValue(target.value)
    if bestName.Text ~= target.name then bestName.Text = target.name end
    local rtext = target.rarity or ""
    if bestRarity.Text ~= rtext then bestRarity.Text = rtext end
    local rc = target.rcolor or C.textMuted
    if bestRarity.TextColor3 ~= rc then bestRarity.TextColor3 = rc end
    if bestValue.Text ~= valStr then bestValue.Text = valStr end
    if target.icon then
        if previewImg.Image ~= target.icon then previewImg.Image = target.icon end
        previewImg.Visible = true
        previewEmoji.Visible = false
    else
        previewImg.Visible = false
        previewEmoji.Visible = true
    end
end

local function updateDropdownRows()
    local eggs = getFilteredEggs()
    for i = 1, DROPDOWN_ROWS do
        local ui = U.rows[i]
        local egg = eggs[i]
        ui.Egg = egg
        if not egg then
            ui.Row.Visible = false
            continue
        end
        ui.Row.Visible = true
        if ui.LastUid ~= egg.uid then
            ui.LastUid = egg.uid
            ui.NameLabel.Text = egg.name
            ui.ValueLabel.Text = fmtValue(egg.value)
            local rtext = egg.rarity or ""
            ui.RarityLabel.Text = rtext
            ui.RarityLabel.TextColor3 = egg.rcolor or C.textMuted
            if egg.icon and egg.icon ~= ui.LastIcon then
                ui.LastIcon = egg.icon
                ui.PreviewImg.Image = egg.icon
                ui.PreviewImg.Visible = true
                ui.PreviewEmoji.Visible = false
            elseif not egg.icon then
                ui.PreviewImg.Visible = false
                ui.PreviewEmoji.Visible = true
            end
        end
        local sel = (HUB.selectedUid == egg.uid)
        if sel then
            ui.Row.BackgroundColor3 = C.cardSel
        else
            ui.Row.BackgroundColor3 = C.card
        end
    end
end

tbl2.startEggProtection = function(heldEggUid)
	if not heldEggUid then
		return
	end

	if tbl4.isProtecting and tbl4.heldEggUid == heldEggUid then
		return
	end
	tbl4.isProtecting = true
	tbl4.heldEggUid = heldEggUid
	tbl4.lastProtectTime = 0

	task.spawn(function()
		while tbl4.isProtecting and tbl4.heldEggUid == heldEggUid do
			tbl4.lastProtectTime = tick()

			local ok, result = pcall(function()
				return tbl.AskFieldEggCarry:InvokeServer({ Uid = heldEggUid })
			end)

			if ok and result == true then
				tbl4.carrying = true
			end

			task.wait(tbl3.PROTECT_INTERVAL)
		end
	end)
end

tbl2.startSpammingCarry = function(heldEggUid)
	if not heldEggUid or tbl4.isStealing then
		return
	end

	if tbl4.carrying and tbl4.heldEggUid == heldEggUid then
		return
	end
	tbl4.isStealing = true
	tbl4.lastSpamTime = 0

	task.spawn(function()
		while true do
			if tbl4.isStealing and not tbl4.carrying then
				local now = tick()

				if not (tbl3.SPAM_INTERVAL <= now - tbl4.lastSpamTime) then
					RunService.Heartbeat:Wait()
					continue
				else
					tbl4.lastSpamTime = now

					local ok, result = pcall(function()
						return tbl.AskFieldEggCarry:InvokeServer({ Uid = heldEggUid })
					end)

					if ok and result == true then
						tbl4.isStealing = false
						tbl4.carrying = true
						tbl4.heldEggUid = heldEggUid

						tbl4.carryLocked = true
						-- FIX: تم حذف السطر المعطوب ;(nil)(...) الذي كان يسبب خطأ
						tbl2.spawnBossForEgg(tbl4.targetEgg or tbl2.getActiveTargetEgg())

						task.spawn(function()
							tbl2.startEggProtection(heldEggUid)
						end)

						task.spawn(function()
							local v16 = tbl2.getGroundPosition()
							local v17 = tbl2.getBossDeliveryPosition(v16)

							if v16 and v17 then
								tbl4.isDelivering = true
								tbl2.clearBeam()
								tbl2.createBeam(v16)

								tbl2.startMove(v17, function()
									tbl4.isDelivering = false
									tbl2.cleanupBossMount()
									tbl2.resetCarryState()
									tbl2.completeGuidedCycle()
								end, true)
							end
						end)

						break
					else
						RunService.Heartbeat:Wait()
						continue
					end
				end
			end

			break
		end

		tbl4.isStealing = false
	end)
end

tbl2.startDelivery = function()
	if tbl4.isDelivering then
		return
	end
	local v16 = tbl2.getGroundPosition()
	local v17 = tbl2.getBossDeliveryPosition(v16)
	if not v16 or not v17 then
		return
	end
	tbl4.isDelivering = true
	tbl2.clearBeam()
	tbl2.createBeam(v16)

	if humanoidRootPart and humanoidRootPart.Parent then
		local vector = Vector3.new(v17.X - humanoidRootPart.Position.X, 0, v17.Z - humanoidRootPart.Position.Z)

		if vector.Magnitude > 0.01 then
			humanoidRootPart.CFrame = CFrame.lookAt(humanoidRootPart.Position, humanoidRootPart.Position + vector.Unit)
		end
	end

	tbl2.startMove(v17, function()
		tbl4.isDelivering = false
		tbl2.cleanupBossMount()
		tbl2.resetCarryState()
		tbl2.completeGuidedCycle()
	end, true)
end

tbl2.setupCharacter = function(arg)
	if tbl2.motorAddedConn then
		tbl2.motorAddedConn:Disconnect()
		tbl2.motorAddedConn = nil
	end

	if tbl2.motorRemovingConn then
		tbl2.motorRemovingConn:Disconnect()
		tbl2.motorRemovingConn = nil
	end

	v = arg
	humanoidRootPart = arg:WaitForChild("HumanoidRootPart")
	humanoid = arg:FindFirstChildOfClass("Humanoid")
	tbl2.motorCache = {}

	tbl2.walk(arg, function(arg2)
		if arg2:IsA("Motor6D") then
			tbl2.motorCache[#tbl2.motorCache + 1] = arg2
		end
	end)

	tbl2.motorAddedConn = arg.DescendantAdded:Connect(function(descendant)
		if descendant:IsA("Motor6D") then
			table.insert(tbl2.motorCache, descendant)
		end
	end)

	tbl2.motorRemovingConn = arg.DescendantRemoving:Connect(function(descendant)
		if not descendant:IsA("Motor6D") then
			return
		end

		for i = #tbl2.motorCache, 1, -1 do
			if tbl2.motorCache[i] == descendant then
				table.remove(tbl2.motorCache, i)
				break
			end
		end
	end)
end

tbl.FieldEggCarry.OnClientEvent:Connect(function(arg)
	if tbl2.newTGRuntime and tbl2.newTGRuntime.running then
		return
	end

	if not arg or not arg.Uid then
		return
	end
	local uid = arg.Uid

	if arg.CarrierUserId == localPlayer.UserId then
		if uid == tbl4.heldEggUid then
			tbl4.carrying = true
			return
		end

		if tbl4.targetEgg and uid == tbl4.targetEgg.Uid then
			tbl4.carrying = true
			tbl4.heldEggUid = uid
			tbl4.carryLocked = true
			tbl4.isStealing = false
			tbl2.spawnBossForEgg(tbl4.targetEgg or tbl2.getActiveTargetEgg())

			task.spawn(function()
				tbl2.startEggProtection(uid)
			end)

			task.spawn(function()
				tbl2.startDelivery()
			end)
		end

		return
	end

	if uid == tbl4.heldEggUid then
		tbl4.carrying = false
		tbl4.carryLocked = false

		if not tbl4.isProtecting then
			task.spawn(function()
				tbl2.startEggProtection(uid)
			end)
		end

		return
	end

	if tbl4.targetEgg and uid == tbl4.targetEgg.Uid then
		tbl4.carrying = false

		if not tbl4.isStealing then
			task.defer(function()
				task.wait(0.02)

				if tbl4.targetEgg and tbl4.targetEgg.Uid == uid and not tbl4.carrying then
					tbl2.startSpammingCarry(uid)
				end
			end)
		end
	end
end)
